cmake_policy(SET CMP0079 NEW)
cmake_policy(SET CMP0076 NEW)
cmake_policy(SET CMP0118 NEW)
find_package(Protobuf REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS WebSockets Multimedia MultimediaWidgets)
set_target_properties(Qt5::WebSockets Qt5::Multimedia Qt5::MultimediaWidgets PROPERTIES IMPORTED_GLOBAL TRUE)
protobuf_generate_cpp(MISSION_PROTO_SRC MISSION_PROTO_HDR mission3dg/protocol/mission.proto)
qt5_add_resources(MISSION_MODEL_RESOURCES mission3dg/models.qrc)
target_sources(CloudCompare PRIVATE mission3dg/QuadrotorModel.cpp ${MISSION_MODEL_RESOURCES})
install(FILES mission3dg/assets/x500/LICENSE DESTINATION share/cloudcompare/licenses RENAME PX4-X500-LICENSE)
target_sources(CloudCompare PRIVATE mission3dg/MissionController.cpp mission3dg/ControllerActions.cpp mission3dg/MissionController.h mission3dg/ProtocolClient.cpp mission3dg/ProtocolClient.h mission3dg/YamlSettingsPanel.cpp mission3dg/YamlSettingsPanel.h mission3dg/CloudIO.cpp mission3dg/CloudIO.h ${MISSION_PROTO_SRC} ${MISSION_PROTO_HDR})
target_include_directories(CloudCompare PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/mission3dg ${CMAKE_CURRENT_BINARY_DIR} ${Protobuf_INCLUDE_DIRS})
target_link_libraries(CloudCompare Qt5::WebSockets Qt5::Multimedia Qt5::MultimediaWidgets ${Protobuf_LIBRARIES})
target_compile_definitions(CloudCompare PRIVATE THREEDG_ENABLED)
set_property(TARGET CloudCompare PROPERTY CXX_STANDARD 17)
add_executable(mission3dg_tests mission3dg/tests.cpp mission3dg/ProtocolClient.cpp mission3dg/ProtocolClient.h mission3dg/CloudIO.cpp ${MISSION_PROTO_SRC} ${MISSION_PROTO_HDR})
target_include_directories(mission3dg_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/mission3dg ${CMAKE_CURRENT_BINARY_DIR} ${Protobuf_INCLUDE_DIRS})
target_link_libraries(mission3dg_tests Qt5::Core Qt5::WebSockets ${Protobuf_LIBRARIES})
set_target_properties(mission3dg_tests PROPERTIES AUTOMOC ON CXX_STANDARD 17)
